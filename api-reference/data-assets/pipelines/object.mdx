<ResponseField name="id" type="uuid" required>
  Unique identifier for the pipeline.
</ResponseField>

<ResponseField name="name" type="string" required>
  Name of the pipeline. Must be unique within the parent service.
</ResponseField>

<ResponseField name="fullyQualifiedName" type="string">
  Fully qualified name in format `{service}.{pipeline}`.
</ResponseField>

<ResponseField name="displayName" type="string">
  Human-readable display name for the pipeline.
</ResponseField>

<ResponseField name="description" type="string">
  Description of the pipeline in Markdown format.
</ResponseField>

<ResponseField name="pipelineUrl" type="string">
  URL to the pipeline in the source system.
</ResponseField>

<ResponseField name="scheduleInterval" type="string">
  Cron schedule expression for the pipeline.
</ResponseField>

<ResponseField name="startDate" type="datetime">
  Pipeline start date.
</ResponseField>

<ResponseField name="tasks" type="Task[]">
  Tasks/operators in the pipeline.
</ResponseField>

<ResponseField name="service" type="EntityReference" required>
  Reference to the parent PipelineService.
</ResponseField>

<ResponseField name="owners" type="EntityReference[]">
  Owners of the pipeline (users or teams).
</ResponseField>

<ResponseField name="domain" type="EntityReference">
  Domain this pipeline belongs to.
</ResponseField>

<ResponseField name="tags" type="TagLabel[]">
  Tags and classifications applied to this pipeline.
</ResponseField>

<ResponseField name="version" type="number">
  Entity version number, incremented on updates.
</ResponseField>

<ResponseField name="deleted" type="boolean">
  Whether the pipeline has been soft-deleted.
</ResponseField>
