<ResponseField name="id" type="uuid" required>
  Unique identifier for the dashboard.
</ResponseField>

<ResponseField name="name" type="string" required>
  Name of the dashboard. Must be unique within the parent service.
</ResponseField>

<ResponseField name="fullyQualifiedName" type="string">
  Fully qualified name in format `{service}.{dashboard}`.
</ResponseField>

<ResponseField name="displayName" type="string">
  Human-readable display name for the dashboard.
</ResponseField>

<ResponseField name="description" type="string">
  Description of the dashboard in Markdown format.
</ResponseField>

<ResponseField name="dashboardUrl" type="string">
  URL to the dashboard in the source system.
</ResponseField>

<ResponseField name="dashboardType" type="string">
  Type of dashboard (e.g., Report, Dashboard).
</ResponseField>

<ResponseField name="service" type="EntityReference" required>
  Reference to the parent DashboardService.
</ResponseField>

<ResponseField name="charts" type="EntityReference[]">
  Charts contained in this dashboard.
</ResponseField>

<ResponseField name="dataModels" type="EntityReference[]">
  Data models associated with this dashboard.
</ResponseField>

<ResponseField name="owners" type="EntityReference[]">
  Owners of the dashboard (users or teams).
</ResponseField>

<ResponseField name="domain" type="EntityReference">
  Domain this dashboard belongs to.
</ResponseField>

<ResponseField name="tags" type="TagLabel[]">
  Tags and classifications applied to this dashboard.
</ResponseField>

<ResponseField name="version" type="number">
  Entity version number, incremented on updates.
</ResponseField>

<ResponseField name="deleted" type="boolean">
  Whether the dashboard has been soft-deleted.
</ResponseField>
