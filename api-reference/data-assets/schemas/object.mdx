<ResponseField name="id" type="uuid" required>
  Unique identifier for the database schema.
</ResponseField>

<ResponseField name="name" type="string" required>
  Name of the schema. Must be unique within the parent database.
</ResponseField>

<ResponseField name="fullyQualifiedName" type="string">
  Fully qualified name in format `{service}.{database}.{schema}`.
</ResponseField>

<ResponseField name="displayName" type="string">
  Human-readable display name for the schema.
</ResponseField>

<ResponseField name="description" type="string">
  Description of the schema in Markdown format.
</ResponseField>

<ResponseField name="database" type="EntityReference" required>
  Reference to the parent Database.
</ResponseField>

<ResponseField name="service" type="EntityReference">
  Reference to the parent DatabaseService (derived from database).
</ResponseField>

<ResponseField name="owners" type="EntityReference[]">
  Owners of the schema (users or teams). Inherited by all child tables.
</ResponseField>

<ResponseField name="domain" type="EntityReference">
  Domain this schema belongs to. Inherited by all child tables.
</ResponseField>

<ResponseField name="tags" type="TagLabel[]">
  Tags and classifications applied to this schema.
</ResponseField>

<ResponseField name="tables" type="EntityReference[]">
  Child tables within this schema (read-only, populated with `fields` param).
</ResponseField>

<ResponseField name="version" type="number">
  Entity version number, incremented on updates.
</ResponseField>

<ResponseField name="deleted" type="boolean">
  Whether the schema has been soft-deleted.
</ResponseField>
